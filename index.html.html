MIN-MAX SCALING

from sklearn.preprocessing import MinMaxScaler
import pandas as pd

# Reload the original DataFrame to ensure correct columns are available
df = pd.read_csv('/content/netflix_dataset.csv')

scaled = ['views_millions', 'duration_min', 'rating']
scaler = MinMaxScaler()

print(df_scaled.head())

Practical 2 — MySQL

Before using import mysql.connector, you may need to run: pip install mysql-connector-python
import mysql.connector

# Connect to MySQL
conn = mysql.connector.connect(
    host="localhost",
    user="root",
    password="your_password"
)
cursor = conn.cursor()

# Create Database
cursor.execute("CREATE DATABASE IF NOT EXISTS mydatabase")
cursor.execute("USE mydatabase")

# Create Table
cursor.execute("""
CREATE TABLE IF NOT EXISTS mytable (
id INT AUTO_INCREMENT PRIMARY KEY,
name VARCHAR(100),
category VARCHAR(100),
value FLOAT,
description VARCHAR(200)
)
""")
print("Database and table created successfully!")

# CREATE - Insert Data
data = [
("Item1", "Category1", 10.5, "Description 1"),
("Item2", "Category2", 20.5, "Description 2"),
("Item3", "Category3", 30.5, "Description 3")
]

cursor.executemany("""
INSERT INTO mytable (name, category, value, description)
VALUES (%s, %s, %s, %s)
""", data)
conn.commit()
print("Data inserted successfully!")

# READ - Fetch All Data
cursor.execute("SELECT * FROM mytable")
print("All Records:")
for row in cursor.fetchall():
    print(row)

# UPDATE - Modify Data
cursor.execute("""
UPDATE mytable
SET value = 40
WHERE name = 'Item2'
""")
conn.commit()
print("Data updated successfully!")

# DELETE - Delete Data
cursor.execute("""
DELETE FROM mytable
WHERE value > 30
""")
conn.commit()
print("Data deleted successfully!")

# Close Connection
cursor.close()
conn.close()
print("MySQL connection closed!")
